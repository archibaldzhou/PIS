   try{p.publish(id,bytes.length,hash);}catch(StorageProvider.Failure f){if(!f.code.equals("STORAGE_BUSY"))throw f;busy++;continue;}
   try{if(!Arrays.equals(p.read(id,bytes.length,hash,null).bytes(),bytes))throw new AssertionError();success++;}catch(StorageProvider.Failure f){if(!f.code.equals("STORAGE_BUSY"))throw f;readBusy++;}
  }}System.out.println("Original contract reproduction rounds=200 replayPublishBusy="+busy+" replayReadBusy="+readBusy+" exactSuccess="+success);
 }
}
